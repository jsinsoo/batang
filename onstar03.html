import turtle, math

t = turtle.Turtle()
t.speed(0)
t.hideturtle()

def rot(x, y, deg):
    a = math.radians(deg)
    return (x*math.cos(a)-y*math.sin(a),
            x*math.sin(a)+y*math.cos(a))

n = 24
s = 170

for ang in range(0, 360, 18):
    for i in range(n + 1):
        a = s*i/n
        b = s*(n-i)/n
        p1 = rot(a, 0, ang)
        p2 = rot(0, b, ang)
        t.penup()
        t.goto(p1)
        t.pendown()
        t.goto(p2)

turtle.done()

right = {'A': 'C', 'B': 'E', 'C': 'G'}

def inorder(node):
    if node in left:
        inorder(left[node])
    print(node, end=" ")
    if node in right:
        inorder(right[node])

def postorder(node):
    if node in left:
        postorder(left[node])
    if node in right:
        postorder(right[node])
    print(node, end=" ")

inorder("A")
print("후위", end=" ")
